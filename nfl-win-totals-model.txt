  const defaultWeights = {
    quarterback:24, defense:24, offensiveLine:11,
    skill:11, coaching:12, prior:18

  const clamp = (x, low, high) => Math.max(low, Math.min(high, x));
  const rankZ = rank => (16.5 - rank) / 9.2331;
  const impliedProbability = odds => odds < 0 ? -odds / (-odds + 100) : 100 / (odds + 100);
  const profitPer100 = odds => odds < 0 ? 10000 / -odds : odds;

  function normalCDF(x) {
    const t = 1 / (1 + .2316419 * Math.abs(x));
    const d = .3989423 * Math.exp(-x * x / 2);
    let p = 1 - d * t * (.319381530 + t * (-.356563782 + t * (1.781477937 + t * (-1.821255978 + t * 1.330274429))));
    return x >= 0 ? p : 1 - p;
  }

  function fairAmericanOdds(p) {
    if (p <= 0 || p >= 1) return null;
    return Math.round(p >= .5 ? -100 * p / (1 - p) : 100 * (1 - p) / p);
  }

  function calculate(options = {}) {
    const data = (options.teams || teams).map(t => ({...t}));
    const weights = {...defaultWeights, ...(options.weights || {})};
    const seasonSD = options.seasonSD == null ? 2.15 : +options.seasonSD;
    const marketBlend = clamp(options.marketBlend == null ? 0 : +options.marketBlend, 0, .35);
    const priorMean = data.reduce((sum,t) => sum + t.priorWins, 0) / data.length;
    const priorSD = Math.sqrt(data.reduce((sum,t) => sum + (t.priorWins-priorMean)**2, 0)/(data.length-1)) || 1;
    const weightTotal = Object.values(weights).reduce((a,b) => a + b, 0) || 1;

    let scored = data.map(t => ({...t, strength:(
      weights.quarterback * rankZ(t.qb) +
      weights.defense * rankZ(t.defense) +
      weights.offensiveLine * rankZ(t.offensiveLine) +
      weights.skill * rankZ(t.skill) +
      weights.coaching * rankZ(t.coaching) +
      weights.prior * ((t.priorWins-priorMean)/priorSD)
    ) / weightTotal}));

    const meanStrength = scored.reduce((sum,t) => sum + t.strength, 0) / scored.length;
    scored = scored.map(t => ({...t, uncorrectedWins:8.5 + 2.05*(t.strength-meanStrength)}));
    const leagueCorrection = (272-scored.reduce((sum,t) => sum+t.uncorrectedWins,0))/scored.length;

    return scored.map(t => {
      const expectedWins = clamp((t.uncorrectedWins+leagueCorrection)*(1-marketBlend)+t.line*marketBlend,1.5,14.5);
      const winSD = seasonSD*t.uncertainty;
      const overProbability = 1-normalCDF((t.line-expectedWins)/winSD);
      const underProbability = 1-overProbability;
      const sides = [
        {side:'Over', odds:t.overOdds, probability:overProbability},
        {side:'Under', odds:t.underOdds, probability:underProbability}
      ].map(x => {
        const breakEven = impliedProbability(x.odds);
        const ev = x.probability*profitPer100(x.odds)-(1-x.probability)*100;
        return {...x,breakEven,probabilityEdge:x.probability-breakEven,evPer100:ev,roi:ev/100,fairOdds:fairAmericanOdds(x.probability)};
      });
      const bestBet = sides.reduce((best,x) => x.evPer100>best.evPer100?x:best);
      return {...t,expectedWins,winSD,overProbability,underProbability,sides,bestBet};
    }).sort((a,b) => b.bestBet.evPer100-a.bestBet.evPer100);
  }

  return {teams, defaultWeights, calculate, impliedProbability, fairAmericanOdds};
